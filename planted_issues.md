## 6. Verification and Audit Methods

Each planted issue can be verified deterministically via SQL queries against `plm_mock.db` or via the `PLMBackend` analytics interface:

```sql
-- 1. Detect Obsolete Parts in Released Assemblies
SELECT p_parent.part_number AS parent, p_parent.name AS parent_name,
       p_child.part_number AS obsolete_child, p_child.name AS child_name,
       b.quantity, b.notes
FROM bom_links b
JOIN part_versions v_parent ON b.parent_version_id = v_parent.id
JOIN part_masters p_parent ON v_parent.master_id = p_parent.id
JOIN part_masters p_child ON b.child_master_id = p_child.id
JOIN part_versions v_child ON v_child.master_id = p_child.id
WHERE v_parent.lifecycle_state = 'RELEASED'
  AND v_child.lifecycle_state = 'OBSOLETE'
  AND v_child.id = (
      SELECT id FROM part_versions WHERE master_id = p_child.id ORDER BY revision DESC, iteration DESC LIMIT 1
  );

-- 2. Detect BUY Parts with No Supplier in AML
SELECT m.part_number, m.name, v.material
FROM part_masters m
JOIN part_versions v ON m.id = v.master_id
WHERE v.make_buy = 'BUY'
  AND m.id NOT IN (SELECT part_master_id FROM aml_links)
  AND v.id = (
      SELECT id FROM part_versions WHERE master_id = m.id ORDER BY revision DESC, iteration DESC LIMIT 1
  );

-- 3. Detect EBOM vs MBOM Mismatches
SELECT ebom.parent_part_number, ebom.child_part_number,
       ebom.quantity AS ebom_qty, mbom.quantity AS mbom_qty,
       ebom.notes, mbom.notes
FROM (SELECT b.*, pm_p.part_number AS parent_part_number, pm_c.part_number AS child_part_number
      FROM bom_links b
      JOIN part_versions pv ON b.parent_version_id = pv.id
      JOIN part_masters pm_p ON pv.master_id = pm_p.id
      JOIN part_masters pm_c ON b.child_master_id = pm_c.id
      WHERE b.bom_type = 'EBOM') ebom
FULL OUTER JOIN
     (SELECT b.*, pm_p.part_number AS parent_part_number, pm_c.part_number AS child_part_number
      FROM bom_links b
      JOIN part_versions pv ON b.parent_version_id = pv.id
      JOIN part_masters pm_p ON pv.master_id = pm_p.id
      JOIN part_masters pm_c ON b.child_master_id = pm_c.id
      WHERE b.bom_type = 'MBOM') mbom
ON ebom.parent_part_number = mbom.parent_part_number
   AND ebom.child_part_number = mbom.child_part_number
WHERE ebom.quantity IS NULL OR mbom.quantity IS NULL OR ebom.quantity != mbom.quantity;

-- 4. Detect RoHS Hazardous Material Risks
SELECT m.part_number, m.name, v.material
FROM part_masters m
JOIN part_versions v ON m.id = v.master_id
WHERE (v.material LIKE '%Lead%' OR v.material LIKE '%Cadmium%'
       OR v.material LIKE '%Hexavalent%' OR v.material LIKE '%Mercury%'
       OR v.material LIKE '%Deca-BDE%' OR v.material LIKE '%DEHP%')
  AND v.id = (
      SELECT id FROM part_versions WHERE master_id = m.id ORDER BY revision DESC, iteration DESC LIMIT 1
  );
```
